cmake_minimum_required(VERSION 3.16)
project(alpha_gold_model_check LANGUAGES CXX)
find_package(Geant4 REQUIRED)
include(${Geant4_USE_FILE})
add_executable(alpha_gold_geant4 main.cc)
target_compile_features(alpha_gold_geant4 PRIVATE cxx_std_17)
target_compile_options(alpha_gold_geant4 PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
target_link_libraries(alpha_gold_geant4 PRIVATE ${Geant4_LIBRARIES})

include(CTest)
if(BUILD_TESTING)
    add_executable(classical_numerics ../analysis/test_numerics.cpp)
    target_compile_features(classical_numerics PRIVATE cxx_std_17)
    add_test(NAME classical_numerics COMMAND classical_numerics)
    add_test(NAME matched_models COMMAND alpha_gold_geant4
             ${CMAKE_CURRENT_SOURCE_DIR}/.. ${CMAKE_CURRENT_BINARY_DIR}/geant4_model_check.root)
endif()

add_executable(trajectory_check trajectory_check.cc classical_bridge.cc)
target_compile_features(trajectory_check PRIVATE cxx_std_17)
target_link_libraries(trajectory_check PRIVATE ${Geant4_LIBRARIES})
if(BUILD_TESTING)
    add_test(NAME trajectory_check COMMAND trajectory_check ${CMAKE_CURRENT_BINARY_DIR}/trajectory_check.csv)
endif()

add_executable(nuclear_physics_benchmark physics_benchmark.cc classical_bridge.cc)
target_compile_features(nuclear_physics_benchmark PRIVATE cxx_std_17)
target_link_libraries(nuclear_physics_benchmark PRIVATE ${Geant4_LIBRARIES})
if(BUILD_TESTING)
    add_test(NAME nuclear_physics_benchmark COMMAND nuclear_physics_benchmark ${CMAKE_CURRENT_BINARY_DIR}/nuclear_physics_benchmark.csv)
endif()

add_executable(nuclear_scattering ../nuclear_scattering.cpp)
target_compile_features(nuclear_scattering PRIVATE cxx_std_17)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
    add_test(NAME output_failures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../analysis/test_output_failures.py $<TARGET_FILE:nuclear_scattering>)
endif()

    include(${Geant4_USE_FILE})
    add_executable(nucleus_viewer nucleus_viewer.cc)
    target_compile_features(nucleus_viewer PRIVATE cxx_std_17)
    target_link_libraries(nucleus_viewer PRIVATE ${Geant4_LIBRARIES})
endif()
